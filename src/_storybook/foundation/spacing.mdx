export const SpacingItem = ({ values, tokenData, tokenPrefix }) => {
  const [spacingValues, setSpacingValues] = React.useState({});
  const [isLoaded, setIsLoaded] = React.useState(false);

  React.useEffect(() => {
    const timer = setTimeout(() => {
      const valueObj = {};
      values.forEach(spacingName => {
        valueObj[spacingName] = tokenData[spacingName];
      });
      setSpacingValues(valueObj);
      setIsLoaded(true);
    }, 0);

    return () => clearTimeout(timer);
  }, [values, tokenData]);

  return (
    <div className="spacing-item">
      <div className="spacing-item__container">
        {values.map((spacingName) => (
          <div key={spacingName} className="spacing-item__example">
            <div className="spacing-item__info-container">
              <div className="spacing-item__sample-wrapper">
                <div
                  className="spacing-item__sample"
                  style={{ width: tokenData[spacingName], height: tokenData[spacingName], width: tokenData[spacingName] }}
                />
              </div>
              <div className="spacing-item__details">
                <div>{spacingName}</div>
                <div className="spacing-item__value">{isLoaded ? spacingValues[spacingName] : 'Loading...'}</div>
                <div className="spacing-item__variable-name">--col-{tokenPrefix}-size-{spacingName}</div>
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};
