import React from 'react';
import { Meta, Title, Subtitle, Source } from '@storybook/blocks';

<Meta title="Foundation/Depth" />

export const depth = {
  'box-shadow': {
    xs: '0 1px 2px 0 rgba(var(--col-theme-black-rgb), .15)',
    s: '0 2px 4px 0 rgba(var(--col-theme-black-rgb), .15)',
    m: '0 4px 8px 0 rgba(var(--col-theme-black-rgb), .15)',
    l: '0 8px 12px 0 rgba(var(--col-theme-black-rgb), .15)',
    xl: '0 12px 16px 0 rgba(var(--col-theme-black-rgb), .15)',
    'xs-inverse': '0 -1px 2px 0 rgba(var(--col-theme-black-rgb), .15)',
    's-inverse': '0 -2px 4px 0 rgba(var(--col-theme-black-rgb), .15)',
    'm-inverse': '0 -4px 8px 0 rgba(var(--col-theme-black-rgb), .15)',
    'l-inverse': '0 -8px 12px 0 rgba(var(--col-theme-black-rgb), .15)',
    'xl-large-inverse': '0 -12px 16px 0 rgba(var(--col-theme-black-rgb), .15)',
    active: '0 0 0 2px var(--col-theme-focus), inset 0 1px 2px 0 rgba(var(--col-theme-black-rgb), .15)',
    focus: '0 0 0 2px var(--col-theme-focus)',
    inset: 'inset 0 1px 2px 0 rgba(var(--col-theme-black-rgb), .15)',
    plain: '0, 0, 0, 0 rgba(var(--col-theme-black-rgb), .15)',
  },
};

<div className="col-doc__wrapper">
  <div className="col-doc__container">
    <Title>Depth</Title>
    <div className="col-intro-text">
      The depth system provides box-shadow tokens that create visual hierarchy and spatial relationships through elevation in the interface. These tokens allow components to appear raised from or recessed into the UI surface, adding dimension and focus while maintaining a cohesive visual language.
    </div>

    <DepthItem
      values={Object.keys(depth['box-shadow'])}
    />
  </div>
</div>

export const DepthItem = ({ values }) => {
  const [depthValues, setDepthValues] = React.useState({});
  const [isLoaded, setIsLoaded] = React.useState(false);

  React.useEffect(() => {
    const timer = setTimeout(() => {
      const valueObj = {};
      values.forEach(depthName => {
        valueObj[depthName] = depth['box-shadow'][depthName];
      });
      setDepthValues(valueObj);
      setIsLoaded(true);
    }, 0);

    return () => clearTimeout(timer);
  }, [values]);

  const isInverse = (name) => name.includes('inverse');

  return (
    <div className="depth-item">
      <div className="depth-item__container">
        {values.map((depthName) => (
          <div key={depthName} className="radius-item__example">
            <div className="radius-item__info-container">
              <div className="radius-item__sample-wrapper">
                <div
                  className={`radius-item__sample depth-sample ${isInverse(depthName) ? 'depth-inverse-sample' : ''}`}
                  style={{ boxShadow: depthValues[depthName] }}
                />
              </div>
              <div className="radius-item__details">
                <div>{depthName}</div>
                <div className="radius-item__value">{isLoaded ? depthValues[depthName] : 'Loading...'}</div>
                <div className="radius-item__variable-name">--col-depth-box-shadow-{depthName}</div>
              </div>
            </div>
            <div className="radius-item__code-container">
              <Source
                dark
                format="dedent"
                language="css"
                code={`box-shadow: var(--col-depth-box-shadow-${depthName});`}
              />
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};
