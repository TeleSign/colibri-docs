## Installation

First, install the required packages:

```bash
npm install @tls-ds/colibri @tls-ds/colibri-icons
```

## Basic Setup

### Component Registration

Create a Colibri plugin file (e.g., `src/plugins/colibri.ts`):

```typescript
import { registerColibriComponents, ColButton, ColTextField } from '@tls-ds/colibri';
import { ColIcon } from '@tls-ds/colibri-icons';

// Register components
registerColibriComponents([
  ColIcon,
  ColButton,
  ColTextField
]);

// Vue plugin
export default {
  install() {
    // Plugin initialization if needed
  }
};
```

Register the plugin in your main file:

```typescript
// src/main.ts
import { createApp } from 'vue';
import App from './App.vue';
import ColibriPlugin from './plugins/colibri';

const app = createApp(App);
app.use(ColibriPlugin);
app.mount('#app');
```

### Theme Setup

Create a theme provider component:

```vue
<!-- src/components/ThemeProvider.vue -->
<template>
  <div :data-theme="theme">
    <slot></slot>
  </div>
</template>

<script lang="ts">
import { defineComponent } from 'vue';

export default defineComponent({
  name: 'ThemeProvider',
  props: {
    theme: {
      type: String,
      default: 'light',
      validator: (value: string) => ['light', 'dark', 'system'].includes(value)
    }
  }
});
</script>
```

## Component Usage

### Basic Component Usage

```vue
<template>
  <div>
    <!-- Button Component -->
    <col-button
      color="primary"
      variant="contained"
      @click="handleClick"
    >
      Click Me
    </col-button>

    <!-- Icon Component -->
    <col-icon
      name="user"
      size="24px"
      color="#000000"
    />

    <!-- Text Field Component -->
    <col-text-field
      label="Username"
      name="username"
      :value="username"
      @input="handleInput"
    >
    </col-text-field>
  </div>
</template>

<script lang="ts">
import { defineComponent, ref } from 'vue';

export default defineComponent({
  name: 'MyComponent',
  setup() {
    const username = ref('');

    const handleClick = () => {
      console.log('Button clicked!');
    };

    const handleInput = (event: Event) => {
      const target = event.target as HTMLInputElement;
      console.log('Value:', target.value);
    };

    return {
      username,
      handleClick,
      handleInput
    };
  }
});
</script>
```

### Form Handling

```vue
<template>
  <form @submit.prevent="handleSubmit">
    <col-text-field
      :value="formData.username"
      @input="formData.username = $event.target.value"
      name="username"
      label="Username"
      required
    />
    
    <col-text-field
      :value="formData.password"
      @input="formData.password = $event.target.value"
      name="password"
      type="password"
      label="Password"
      required
    />

    <col-button type="submit">
      Login
    </col-button>
  </form>
</template>

<script lang="ts">
import { defineComponent, reactive } from 'vue';

export default defineComponent({
  name: 'LoginForm',
  setup() {
    const formData = reactive({
      username: '',
      password: ''
    });

    const handleSubmit = () => {
      console.log('Form data:', formData);
    };

    return {
      formData,
      handleSubmit
    };
  }
});
</script>
```

## Theme Configuration

### Theme Provider Usage

```vue
<!-- App.vue -->
<template>
  <theme-provider :theme="currentTheme">
    <your-components />
  </theme-provider>
</template>

<script lang="ts">
import { defineComponent, ref } from 'vue';
import ThemeProvider from './components/ThemeProvider.vue';

export default defineComponent({
  components: {
    ThemeProvider
  },
  setup() {
    const currentTheme = ref('light');
    return { currentTheme };
  }
});
</script>
```

### Dynamic Theme Switching

```vue
<template>
  <theme-provider :theme="theme">
    <col-button @click="toggleTheme">
      Toggle Theme
    </col-button>
    <your-components />
  </theme-provider>
</template>

<script lang="ts">
import { defineComponent, ref } from 'vue';

export default defineComponent({
  setup() {
    const theme = ref<'light' | 'dark' | 'system'>('light');

    const toggleTheme = () => {
      theme.value = theme.value === 'light' ? 'dark' : 'light';
    };

    return {
      theme,
      toggleTheme
    };
  }
});
</script>
```


### Using with Composition API

```typescript
import { defineComponent, ref } from 'vue';
import type { COLORS, BUTTON_VARIANTS } from '@tls-ds/colibri';

export default defineComponent({
  setup() {
    const buttonColor = ref<COLORS>('primary');
    const buttonVariant = ref<BUTTON_VARIANTS>('contained');

    return {
      buttonColor,
      buttonVariant
    };
  }
});
```

## Best Practices

1. **Component Registration**
   - Register components at the application level
   - Use the Vue plugin system for setup

2. **Reactivity**
   - Use `ref` or `reactive` for state management
   - Watch for changes using `watch` or `watchEffect`

3. **Event Handling**
   - Use `@event` syntax for event binding
   - Handle events in setup or methods

4. **Performance**
   - Use `v-show` instead of `v-if` when appropriate
   - Implement proper memoization with `computed`
