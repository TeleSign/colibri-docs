export const ColorItem = ({ name, values, prefix }) => {
  const [colorValues, setColorValues] = React.useState({});
  const [isLoaded, setIsLoaded] = React.useState(false);

  React.useEffect(() => {
    const timer = setTimeout(() => {
      const valueObj = {};
      values.forEach(colorName => {
        valueObj[colorName] = getVariableValue(colorName, prefix);
      });
      setColorValues(valueObj);
      setIsLoaded(true);
    }, 0);

    return () => clearTimeout(timer);
  }, [values, prefix]);

  return (
    <div style={{
      display: 'flex',
      flexDirection: 'column',
      alignItems: 'flex-start',
      marginBottom: '2rem'
    }}>
      <div style={{ display: 'block', marginBottom: '0.5rem' }}>{capitalize(name)}</div>
      <div style={{
        display: 'flex',
        flexWrap: 'wrap',
        gap: '0.5rem',
        width: '100%'
      }}>
        {values.map((colorName) => (
          <div
            key={colorName}
            style={{
              display: 'flex',
              flex: '1',
              flexDirection: 'column',
              textAlign: 'center',
              minWidth: '120px',
              maxWidth: '180px'
            }}
          >
            <div
              style={{
                display: 'flex',
                width: '100%',
                height: '3.5rem',
                border: '1px solid var(--col-border-color, #ccc)',
                marginBottom: '0.25rem',
                backgroundColor: `var(--col-${prefix}-${colorName})`
              }}
            ></div>
            <p style={{
              fontSize: '0.75rem',
              fontWeight: 'bold',
              margin: '0.125rem 0'
            }}>{`--col-${prefix}-${colorName}`}</p>
            <p style={{
              fontSize: '0.75rem',
              color: 'var(--col-text-secondary, #666)',
              margin: '0'
            }}>
              {isLoaded ? colorValues[colorName] : 'Loading...'}
            </p>
          </div>
        ))}
      </div>
    </div>
  );
};
